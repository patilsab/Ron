Here is a complete, automated installation script designed for Arch Linux.

It handles everything end-to-end:

1. **Bootstraps prerequisites** (`git`, `base-devel`, and `yay` AUR helper if not already installed).
2. **Clones the repository** `hacktheclown/arch-clown-dotfiles`.
3. **Installs all standard dependencies** required by the dotfile components (Window managers, bar, terminal, compositor, app launcher, notification daemon, wallpaper, fonts, audio, and shell tools).
4. **Detects any package lists** (like `packages.txt` or `pkglist.txt`) inside the repo and installs them automatically.
5. **Safely backs up** any existing config files before deploying the new dotfiles.
6. **Copies/links configs, scripts, fonts, and assets** into their appropriate home directories and sets executable permissions.

---

### Installation Script (`install-arch-clown.sh`)

Save the script to a file and run it:

```bash
#!/usr/bin/env bash
# ==============================================================================
# Arch Clown Dotfiles - Automated Installer
# Repository: https://github.com/hacktheclown/arch-clown-dotfiles
# ==============================================================================

set -euo pipefail

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
BLUE='\033[0;34m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

REPO_URL="https://github.com/hacktheclown/arch-clown-dotfiles.git"
TARGET_DIR="${HOME}/.cache/arch-clown-dotfiles"
BACKUP_DIR="${HOME}/.config-backup-$(date +%Y%m%d_%H%M%S)"

log_info() { echo -e "${BLUE}[INFO]${NC} $1"; }
log_success() { echo -e "${GREEN}[OK]${NC} $1"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_err() { echo -e "${RED}[ERROR]${NC} $1"; }

# ------------------------------------------------------------------------------
# 1. System & Sudo Verification
# ------------------------------------------------------------------------------
if [ "$EUID" -eq 0 ]; then
    log_err "Please run this script as a normal user (with sudo privileges), NOT as root."
    exit 1
fi

log_info "Synchronizing package databases..."
sudo pacman -Sy --noconfirm --needed base-devel git

# ------------------------------------------------------------------------------
# 2. AUR Helper Setup (yay)
# ------------------------------------------------------------------------------
if ! command -v yay &> /dev/null; then
    log_warn "'yay' AUR helper not found. Installing yay automatically..."
    TEMP_DIR=$(mktemp -d)
    git clone https://aur.archlinux.org/yay.git "$TEMP_DIR/yay"
    (cd "$TEMP_DIR/yay" && makepkg -si --noconfirm)
    rm -rf "$TEMP_DIR"
    log_success "yay installed successfully."
else
    log_info "'yay' is already installed."
fi

# ------------------------------------------------------------------------------
# 3. Core Arch Rice Dependencies
# ------------------------------------------------------------------------------
log_info "Installing core dependencies and common rice components..."

DEPENDENCIES=(
    # Window Managers / Compositors / Bars
    bspwm sxhkd i3-wm hyprland polybar waybar picom
    # Launchers, Terminals, Notifiers
    rofi wofi kitty alacritty dunst feh swww
    # Shell & Utilities
    zsh starship fastfetch neofetch fish
    eza bat ripgrep fzf zoxide brightnessctl
    # Audio & System Controls
    pipewire pipewire-pulse wireplumber pamixer pavucontrol playerctl
    # Theming & Fonts
    ttf-jetbrains-mono-nerd ttf-font-awesome noto-fonts noto-fonts-emoji
    papirus-icon-theme lxappearance qt5ct qt6ct
)

for pkg in "${DEPENDENCIES[@]}"; do
    if ! pacman -Qi "$pkg" &> /dev/null && ! yay -Qi "$pkg" &> /dev/null; then
        log_info "Installing package: $pkg"
        yay -S --noconfirm --needed "$pkg" || log_warn "Optional package $pkg could not be installed (skipping)."
    fi
done

# ------------------------------------------------------------------------------
# 4. Clone / Pull the Dotfiles Repository
# ------------------------------------------------------------------------------
if [ -d "$TARGET_DIR/.git" ]; then
    log_info "Updating existing repository in $TARGET_DIR..."
    git -C "$TARGET_DIR" pull
else
    log_info "Cloning $REPO_URL into $TARGET_DIR..."
    rm -rf "$TARGET_DIR"
    git clone "$REPO_URL" "$TARGET_DIR"
fi

# ------------------------------------------------------------------------------
# 5. Check & Install Any Repository-Specific Package Lists
# ------------------------------------------------------------------------------
find "$TARGET_DIR" -maxdepth 2 -type f \( -name "*packages*.txt" -o -name "*pkglist*.txt" -o -name "*deps*.txt" \) | while read -r pkgfile; do
    log_info "Found package list: $(basename "$pkgfile"). Installing dependencies..."
    while IFS= read -r line || [ -n "$line" ]; do
        # Ignore comments and empty lines
        clean_pkg=$(echo "$line" | sed -e 's/#.*//' -e 's/[[:space:]]*//g')
        if [ -n "$clean_pkg" ]; then
            yay -S --noconfirm --needed "$clean_pkg" || log_warn "Could not install $clean_pkg"
        fi
    done < "$pkgfile"
done

# ------------------------------------------------------------------------------
# 6. Backup Existing Configs and Deploy
# ------------------------------------------------------------------------------
log_info "Backing up existing configurations to $BACKUP_DIR..."
mkdir -p "$BACKUP_DIR"
mkdir -p "${HOME}/.config" "${HOME}/.local/bin" "${HOME}/.local/share/fonts"

# Deploy .config contents
if [ -d "$TARGET_DIR/.config" ]; then
    for item in "$TARGET_DIR/.config"/*; do
        target_name=$(basename "$item")
        if [ -e "${HOME}/.config/$target_name" ]; then
            cp -r "${HOME}/.config/$target_name" "$BACKUP_DIR/"
        fi
        log_info "Deploying config: $target_name"
        cp -r "$item" "${HOME}/.config/"
    done
fi

# Deploy root-level dotfiles (e.g. .zshrc, .bashrc, .xinitrc, etc.)
find "$TARGET_DIR" -maxdepth 1 -name ".*" -not -name "." -not -name ".." -not -name ".git" -not -name ".github" | while read -r dotfile; do
    fname=$(basename "$dotfile")
    if [ "$fname" != ".config" ]; then
        if [ -e "${HOME}/$fname" ]; then
            cp -r "${HOME}/$fname" "$BACKUP_DIR/"
        fi
        log_info "Deploying file: $fname"
        cp -r "$dotfile" "${HOME}/"
    fi
done

# Deploy custom scripts or fonts if present
if [ -d "$TARGET_DIR/.local/bin" ] || [ -d "$TARGET_DIR/bin" ] || [ -d "$TARGET_DIR/scripts" ]; then
    log_info "Deploying helper scripts..."
    [ -d "$TARGET_DIR/.local/bin" ] && cp -r "$TARGET_DIR/.local/bin/"* "${HOME}/.local/bin/" 2>/dev/null || true
    [ -d "$TARGET_DIR/bin" ] && cp -r "$TARGET_DIR/bin/"* "${HOME}/.local/bin/" 2>/dev/null || true
    [ -d "$TARGET_DIR/scripts" ] && cp -r "$TARGET_DIR/scripts/"* "${HOME}/.local/bin/" 2>/dev/null || true
fi

if [ -d "$TARGET_DIR/fonts" ] || [ -d "$TARGET_DIR/.local/share/fonts" ]; then
    log_info "Installing fonts and updating font cache..."
    [ -d "$TARGET_DIR/fonts" ] && cp -r "$TARGET_DIR/fonts/"* "${HOME}/.local/share/fonts/" 2>/dev/null || true
    [ -d "$TARGET_DIR/.local/share/fonts" ] && cp -r "$TARGET_DIR/.local/share/fonts/"* "${HOME}/.local/share/fonts/" 2>/dev/null || true
    fc-cache -fv &> /dev/null || true
fi

# Ensure executable permissions for scripts and wm configs
find "${HOME}/.config" -type f \( -name "*.sh" -o -name "bspwmrc" -o -name "sxhkdrc" \) -exec chmod +x {} + 2>/dev/null || true
find "${HOME}/.local/bin" -type f -exec chmod +x {} + 2>/dev/null || true

# ------------------------------------------------------------------------------
# 7. Completion
# ------------------------------------------------------------------------------
log_success "Installation complete!"
echo -e "${GREEN}Old configuration backups stored in:${NC} $BACKUP_DIR"
echo -e "${YELLOW}Tip:${NC} Restart your window manager or log out and log back in to apply all changes."

```

---

### How to Run

1. Make the script executable:
```bash
chmod +x install-arch-clown.sh

```


2. Run it as your regular user:
```bash
./install-arch-clown.sh

```
