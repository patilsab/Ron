Here is a fully automated installation and deployment script to replicate the **`hacktheclown/arch-clown-dotfiles`** setup on your Arch Linux system.

---

### Automated Setup Script (`install_clown_dotfiles.sh`)

This script handles updating your system, installing the `yay` AUR helper, deploying core window manager / terminal utilities, backing up existing configs, cloning the repository, and copying the dotfiles into place.

```bash
#!/usr/bin/env bash
# ==============================================================================
# Arch Linux Setup Script for hacktheclown/arch-clown-dotfiles
# ==============================================================================

set -e

REPO_URL="https://github.com/hacktheclown/arch-clown-dotfiles.git"
CLONE_DIR="$HOME/arch-clown-dotfiles"
BACKUP_DIR="$HOME/.config_backup_$(date +%Y%m%d_%H%M%S)"

# Color definitions
GREEN='\033[0;32m'
BLUE='\033[0;34m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
NC='\033[0m'

log() { echo -e "${BLUE}[*]${NC} $1"; }
success() { echo -e "${GREEN}[+]${NC} $1"; }
warn() { echo -e "${YELLOW}[!]${NC} $1"; }
error() { echo -e "${RED}[-]${NC} $1"; exit 1; }

# 1. Root Check
if [[ $EUID -eq 0 ]]; then
   error "Do not run this script as root or with sudo. Run as your regular user."
fi

# 2. System Update & Base Development Packages
log "Updating package database and installing base dependencies..."
sudo pacman -Syu --needed --noconfirm base-devel git curl wget rsync

# 3. Install AUR Helper (yay) if not already installed
if ! command -v yay &> /dev/null; then
    log "yay not found. Installing yay from AUR..."
    TEMP_DIR=$(mktemp -d)
    git clone https://aur.archlinux.org/yay.git "$TEMP_DIR/yay"
    pushd "$TEMP_DIR/yay" > /dev/null
    makepkg -si --noconfirm
    popd > /dev/null
    rm -rf "$TEMP_DIR"
    success "yay installed successfully."
else
    log "yay is already installed."
fi

# 4. Essential Packages & Fonts
log "Installing common window manager, terminal, utility, and font packages..."
PKGS=(
    # Window Managers / Launchers / Bars
    bspwm sxhkd i3-wm polybar rofi picom dunst feh nitrogen
    # Terminal & Shell
    kitty alacritty zsh zsh-autosuggestions zsh-syntax-highlighting starship
    # Audio & Media
    pipewire pipewire-pulse pipewire-alsa wireplumber pamixer pavucontrol
    # File Manager & System Info
    thunar neofetch fastfetch btop htop ranger flameshot xclip
    # Fonts
    ttf-jetbrains-mono-nerd ttf-font-awesome noto-fonts-emoji ttf-fira-code
    # Wallpaper & Display Utilities
    xorg-xrandr xdotool brightnessctl lxappearance papirus-icon-theme
)

yay -S --needed --noconfirm "${PKGS[@]}"

# 5. Clone the Dotfiles Repository
if [[ -d "$CLONE_DIR" ]]; then
    warn "Directory $CLONE_DIR already exists. Pulling latest changes..."
    git -C "$CLONE_DIR" pull
else
    log "Cloning hacktheclown/arch-clown-dotfiles repository..."
    git clone "$REPO_URL" "$CLONE_DIR"
fi

# 6. Backup Existing Configuration
log "Creating backup of existing ~/.config files in $BACKUP_DIR..."
mkdir -p "$BACKUP_DIR"

# 7. Deploy Dotfiles
log "Deploying dotfiles to user directories..."

# Copy ~/.config items if present in the repo
if [[ -d "$CLONE_DIR/.config" ]]; then
    mkdir -p "$HOME/.config"
    cp -r "$CLONE_DIR/.config/"* "$HOME/.config/"
    success "Deployed .config contents to ~/.config"
fi

# Copy any root home files (e.g., .zshrc, .bashrc, .xinitrc, scripts)
shopt -s dotglob nullglob
for item in "$CLONE_DIR"/*; do
    name=$(basename "$item")
    # Skip git repo files and README/license files
    if [[ "$name" =~ ^(\.git|\.github|\.config|README.*|LICENSE.*|\.gitignore)$ ]]; then
        continue
    fi
    
    # If file/dir already exists in $HOME, backup first
    if [[ -e "$HOME/$name" ]]; then
        cp -r "$HOME/$name" "$BACKUP_DIR/"
    fi

    cp -r "$item" "$HOME/"
done
shopt -u dotglob nullglob

# 8. Set Execution Permissions on Scripts & Configs
log "Setting execution permissions on scripts and config autostart files..."
chmod +x "$HOME"/.config/bspwm/bspwmrc 2>/dev/null || true
chmod +x "$HOME"/.config/sxhkd/sxhkdrc 2>/dev/null || true
chmod +x "$HOME"/.config/polybar/launch.sh 2>/dev/null || true
find "$HOME/.config" -type f -name "*.sh" -exec chmod +x {} + 2>/dev/null || true
find "$HOME/.local/bin" -type f -exec chmod +x {} + 2>/dev/null || true

# 9. Set Default Shell to Zsh (Optional Prompt)
if [[ "$SHELL" != *"zsh"* ]]; then
    log "Changing default shell to zsh for $USER..."
    chsh -s "$(which zsh)"
fi

echo ""
echo -e "${GREEN}====================================================${NC}"
echo -e "${GREEN}  Installation & Dotfiles Deployment Complete!     ${NC}"
echo -e "${GREEN}====================================================${NC}"
echo -e "• Previous configurations backed up to: ${YELLOW}$BACKUP_DIR${NC}"
echo -e "• Please log out and log back into your window manager / desktop session."

```

---

### How to Run the Script

1. **Save the script to your Arch Linux machine:**
```bash
nano install_clown_dotfiles.sh

```


*(Paste the script contents above and save with `Ctrl+O`, then exit with `Ctrl+X`)*
2. **Make the script executable:**
```bash
chmod +x install_clown_dotfiles.sh

```


3. **Run the script as your standard user:**
```bash
./install_clown_dotfiles.sh

```


4. **Restart your session:** Log out or reboot your system to apply all window manager keybindings, fonts, and shell configurations.
