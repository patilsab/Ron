```
#!/usr/bin/env bash
#
# install.sh — installer for hacktheclown/arch-clown-dotfiles
# https://github.com/hacktheclown/arch-clown-dotfiles
#
# What it does:
#   1. Clones (or updates) the dotfiles repo.
#   2. Backs up any existing files/dirs it would overwrite.
#   3. Symlinks .config/* into ~/.config, plus .zshrc and .p10k.zsh into $HOME.
#   4. Optionally installs the packages the dotfiles expect (pass --packages).
#
# Usage:
#   ./install.sh              # just link the dotfiles
#   ./install.sh --packages   # also pacman -S the required apps first
#   ./install.sh --dry-run    # show what would happen, change nothing
#
set -euo pipefail

REPO_URL="https://github.com/hacktheclown/arch-clown-dotfiles.git"
CLONE_DIR="${DOTFILES_DIR:-$HOME/.arch-clown-dotfiles}"
BACKUP_DIR="$HOME/.dotfiles-backup/$(date +%Y%m%d-%H%M%S)"
WALLPAPER_DEST="$HOME/Pictures/wallpaper.jpg"

DO_PACKAGES=false
DRY_RUN=false

for arg in "$@"; do
  case "$arg" in
    --packages) DO_PACKAGES=true ;;
    --dry-run)  DRY_RUN=true ;;
    -h|--help)
      grep '^#' "$0" | sed 's/^#//'
      exit 0
      ;;
    *)
      echo "Unknown option: $arg" >&2
      exit 1
      ;;
  esac
done

log()  { printf '\033[1;32m==>\033[0m %s\n' "$1"; }
warn() { printf '\033[1;33m==>\033[0m %s\n' "$1"; }
run()  { if $DRY_RUN; then echo "[dry-run] $*"; else "$@"; fi; }

# --- sanity checks -----------------------------------------------------
if ! command -v git >/dev/null 2>&1; then
  echo "git is required but not installed. Run: sudo pacman -S git" >&2
  exit 1
fi

if [[ -f /etc/os-release ]] && ! grep -qi arch /etc/os-release; then
  warn "This doesn't look like Arch Linux — the dotfiles assume Arch/i3. Continuing anyway."
fi

# --- optional: install packages -----------------------------------------
if $DO_PACKAGES; then
  log "Installing required packages via pacman"
  PACKAGES=(
    i3-wm zsh neovim polybar rofi terminator
    ttf-roboto noto-fonts
  )
  run sudo pacman -Syu --needed "${PACKAGES[@]}"

  if [[ ! -d "$HOME/.oh-my-zsh" ]]; then
    log "Installing oh-my-zsh"
    run sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)" "" --unattended
  fi

  P10K_DIR="${ZSH_CUSTOM:-$HOME/.oh-my-zsh/custom}/themes/powerlevel10k"
  if [[ ! -d "$P10K_DIR" ]]; then
    log "Installing powerlevel10k theme"
    run git clone --depth=1 https://github.com/romkatv/powerlevel10k.git "$P10K_DIR"
  fi

  if ! command -v atuin >/dev/null 2>&1; then
    log "Installing atuin"
    run bash -c "$(curl -fsSL https://setup.atuin.sh)"
  fi
fi

# --- clone or update the dotfiles repo ----------------------------------
if [[ -d "$CLONE_DIR/.git" ]]; then
  log "Dotfiles repo already present at $CLONE_DIR — pulling latest"
  run git -C "$CLONE_DIR" pull --ff-only
else
  log "Cloning $REPO_URL into $CLONE_DIR"
  run git clone "$REPO_URL" "$CLONE_DIR"
fi

if $DRY_RUN && [[ ! -d "$CLONE_DIR" ]]; then
  warn "Dry run and repo not actually cloned yet — skipping link step."
  exit 0
fi

# --- backup + link helper -----------------------------------------------
backup_and_link() {
  local src="$1" dest="$2"

  if [[ -L "$dest" ]]; then
    # Already a symlink — just remove it, no need to back up a symlink.
    run rm -f "$dest"
  elif [[ -e "$dest" ]]; then
    log "Backing up existing $dest -> $BACKUP_DIR/"
    run mkdir -p "$BACKUP_DIR/$(dirname "${dest#$HOME/}")"
    run mv "$dest" "$BACKUP_DIR/${dest#$HOME/}"
  fi

  run mkdir -p "$(dirname "$dest")"
  log "Linking $dest -> $src"
  run ln -s "$src" "$dest"
}

# --- link ~/.config/* ----------------------------------------------------
if [[ -d "$CLONE_DIR/.config" ]]; then
  mkdir -p "$HOME/.config"
  for item in "$CLONE_DIR"/.config/*; do
    name="$(basename "$item")"
    backup_and_link "$item" "$HOME/.config/$name"
  done
else
  warn "No .config directory found in the repo — skipping."
fi

# --- link top-level dotfiles --------------------------------------------
[[ -f "$CLONE_DIR/.zshrc"   ]] && backup_and_link "$CLONE_DIR/.zshrc"   "$HOME/.zshrc"
[[ -f "$CLONE_DIR/.p10k.zsh" ]] && backup_and_link "$CLONE_DIR/.p10k.zsh" "$HOME/.p10k.zsh"

# --- wallpaper (not meant to live in ~/.config, per README) -------------
if [[ -f "$CLONE_DIR/wallpaper.jpg" ]]; then
  run mkdir -p "$(dirname "$WALLPAPER_DEST")"
  log "Copying wallpaper to $WALLPAPER_DEST"
  run cp -f "$CLONE_DIR/wallpaper.jpg" "$WALLPAPER_DEST"
fi

# --- change zsh's default shell if needed --------------------------------
if [[ "$SHELL" != *zsh* ]] && command -v zsh >/dev/null 2>&1; then
  warn "Your default shell isn't zsh. Run: chsh -s \$(which zsh)"
fi

log "Done."
if [[ -d "$BACKUP_DIR" ]]; then
  echo "  Backups of anything overwritten: $BACKUP_DIR"
fi
echo "  Repo cloned to: $CLONE_DIR"
echo
warn "Per the repo's README: grep for 'changeme' and 'HTC_NOTE' across the"
warn "configs to find machine-specific values you still need to edit"
warn "(e.g. grep -rn 'changeme\\|HTC_NOTE' \"$CLONE_DIR/.config\")."
```

What it does:
- Clones the repo to `~/.arch-clown-dotfiles` (pulls latest if it's already there)
- Backs up anything it would overwrite to `~/.dotfiles-backup/<timestamp>/` before touching it — nothing gets silently clobbered
- Symlinks every folder under `.config/` into `~/.config/`, plus `.zshrc` and `.p10k.zsh` into `$HOME`
- Copies `wallpaper.jpg` to `~/Pictures/` (the README notes it isn't meant to live under `~/.config`)
- Reminds you at the end to `grep -rn 'changeme\|HTC_NOTE'` across the configs, since the author left machine-specific placeholders you need to edit by hand — there's no automated templating in this repo (the author mentions an Ansible role for that, not yet published)

One thing to flag: since I couldn't browse into `.config/` directly, the script links whatever subfolders it finds there generically rather than hardcoding names like `i3` or `polybar` — so it'll work correctly regardless of the exact contents.
