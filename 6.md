Here’s a complete **installation script** for the *arch-clown-dotfiles* setup on Arch Linux.

It installs the required packages, sets up zsh + oh-my-zsh + powerlevel10k + plugins + atuin, clones the dotfiles, copies them into place, replaces the `changeme` placeholders with your username, and does some basic post-install steps.

### Important Warnings
- This is for **Arch Linux** only.
- Run it on a relatively fresh system (or at least one where you don’t mind overwriting existing configs).
- It will **overwrite** existing configs in `~/.config`, `~/.zshrc`, `~/.p10k.zsh`, etc.
- Review the script before running it.
- The original author is working on a proper Ansible role — this is a practical interim solution.

### Install Script

```bash
#!/usr/bin/env bash
set -euo pipefail

# ============================================================
#  Hack the Clown - Arch Dotfiles Installer
#  https://github.com/hacktheclown/arch-clown-dotfiles
# ============================================================

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

info()  { echo -e "${BLUE}[INFO]${NC} $1"; }
ok()    { echo -e "${GREEN}[OK]${NC} $1"; }
warn()  { echo -e "${YELLOW}[WARN]${NC} $1"; }
error() { echo -e "${RED}[ERROR]${NC} $1"; exit 1; }

# ---------- Checks ----------
[[ $EUID -eq 0 ]] && error "Do NOT run this script as root. Run as your normal user."

if ! command -v pacman &>/dev/null; then
    error "This script only works on Arch Linux (or Arch-based)."
fi

USERNAME=$(whoami)
HOME_DIR="$HOME"
DOTFILES_REPO="https://github.com/hacktheclown/arch-clown-dotfiles.git"
DOTFILES_DIR="$HOME/.dotfiles-arch-clown"

echo
info "Installing for user: $USERNAME"
info "Home directory: $HOME_DIR"
echo

read -rp "Continue? [y/N] " confirm
[[ "$confirm" =~ ^[Yy]$ ]] || exit 0

# ---------- 1. Install packages ----------
info "Updating system..."
sudo pacman -Syu --noconfirm

info "Installing official packages..."
sudo pacman -S --needed --noconfirm \
    i3-wm i3status i3lock \
    polybar \
    rofi \
    terminator \
    picom \
    neovim \
    zsh \
    fzf \
    feh \
    xsettingsd \
    dex \
    git \
    curl \
    wget \
    base-devel \
    ttf-roboto \
    ttf-martian-mono-nerd \
    noto-fonts \
    ttf-nerd-fonts-symbols \
    ttf-nerd-fonts-symbols-mono

# Optional but recommended
sudo pacman -S --needed --noconfirm \
    networkmanager \
    network-manager-applet \
    dunst \
    xss-lock \
    xorg-xrandr \
    xorg-xsetroot \
    xorg-xprop || true

# ---------- 2. AUR helper (yay) ----------
if ! command -v yay &>/dev/null; then
    info "Installing yay (AUR helper)..."
    cd /tmp
    git clone https://aur.archlinux.org/yay.git
    cd yay
    makepkg -si --noconfirm
    cd ~
    rm -rf /tmp/yay
fi

info "Installing AUR packages..."
yay -S --needed --noconfirm \
    atuin \
    ttf-noto-nerd || warn "Some AUR packages failed (you can install them later)"

# ---------- 3. Oh My Zsh + Powerlevel10k + plugins ----------
if [[ ! -d "$HOME/.oh-my-zsh" ]]; then
    info "Installing Oh My Zsh..."
    sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)" "" --unattended
fi

ZSH_CUSTOM="${ZSH_CUSTOM:-$HOME/.oh-my-zsh/custom}"

info "Installing Powerlevel10k..."
if [[ ! -d "$ZSH_CUSTOM/themes/powerlevel10k" ]]; then
    git clone --depth=1 https://github.com/romkatv/powerlevel10k.git \
        "$ZSH_CUSTOM/themes/powerlevel10k"
fi

info "Installing zsh plugins..."
[[ ! -d "$ZSH_CUSTOM/plugins/zsh-autosuggestions" ]] && \
    git clone https://github.com/zsh-users/zsh-autosuggestions \
        "$ZSH_CUSTOM/plugins/zsh-autosuggestions"

[[ ! -d "$ZSH_CUSTOM/plugins/zsh-syntax-highlighting" ]] && \
    git clone https://github.com/zsh-users/zsh-syntax-highlighting \
        "$ZSH_CUSTOM/plugins/zsh-syntax-highlighting"

# ---------- 4. Atuin ----------
if ! command -v atuin &>/dev/null; then
    info "Installing Atuin..."
    curl --proto '=https' --tlsv1.2 -LsSf https://setup.atuin.sh | sh
fi

# ---------- 5. Clone & deploy dotfiles ----------
info "Cloning dotfiles..."
if [[ -d "$DOTFILES_DIR" ]]; then
    warn "Dotfiles directory already exists. Pulling latest..."
    git -C "$DOTFILES_DIR" pull || true
else
    git clone "$DOTFILES_REPO" "$DOTFILES_DIR"
fi

info "Deploying configuration files..."

# Backup existing configs
BACKUP_DIR="$HOME/.dotfiles-backup-$(date +%Y%m%d-%H%M%S)"
mkdir -p "$BACKUP_DIR"

backup_and_copy() {
    local src="$1"
    local dest="$2"
    if [[ -e "$dest" ]]; then
        mkdir -p "$(dirname "$BACKUP_DIR/$dest")"
        cp -a "$dest" "$BACKUP_DIR/" 2>/dev/null || true
    fi
    mkdir -p "$(dirname "$dest")"
    cp -a "$src" "$dest"
}

# Main configs
backup_and_copy "$DOTFILES_DIR/.zshrc"           "$HOME/.zshrc"
backup_and_copy "$DOTFILES_DIR/.p10k.zsh"        "$HOME/.p10k.zsh"
backup_and_copy "$DOTFILES_DIR/.config"          "$HOME/.config"

# Wallpaper
mkdir -p "$HOME/Pictures"
cp -a "$DOTFILES_DIR/wallpaper.jpg" "$HOME/Pictures/wallpaper.jpg"

# ---------- 6. Replace placeholders ----------
info "Replacing 'changeme' with your username ($USERNAME)..."

# Replace in all config files
find "$HOME/.config" -type f -exec sed -i "s|/home/changeme|$HOME|g" {} +
find "$HOME/.config" -type f -exec sed -i "s|changeme|$USERNAME|g" {} +

# Also fix the wallpaper path if needed
sed -i "s|/home/changeme/Pictures|$HOME/Pictures|g" "$HOME/.config/i3/config" 2>/dev/null || true

# ---------- 7. Make scripts executable ----------
info "Making scripts executable..."
find "$HOME/.config/i3/scripts" -type f -exec chmod +x {} + 2>/dev/null || true
chmod +x "$HOME/.config/polybar/polybar_launch.sh" 2>/dev/null || true

# ---------- 8. Set zsh as default shell ----------
if [[ "$SHELL" != */zsh ]]; then
    info "Changing default shell to zsh..."
    chsh -s "$(which zsh)"
    ok "Default shell changed to zsh (takes effect on next login)"
fi

# ---------- 9. Final notes ----------
echo
ok "Installation finished!"
echo
echo -e "${YELLOW}=== Next Steps ===${NC}"
echo "1. Log out and log back in (or reboot)."
echo "2. Select the 'i3' session at the login screen."
echo "3. Review the configs — search for remaining placeholders:"
echo "     grep -r 'changeme\|HTC_NOTE' ~/.config ~/.zshrc"
echo "4. Especially check:"
echo "     - Network interface name in polybar (~/.config/polybar/polybar.ini)"
echo "     - DPI settings (rofi, fonts) if you are not on a high-DPI display"
echo "     - Wallpaper path if you moved it"
echo "5. You may want to install a display manager (e.g. lightdm or sddm)"
echo "   if you don't already have one."
echo
echo "Backups of your old configs (if any) are in:"
echo "  $BACKUP_DIR"
echo
echo "Enjoy the minimal black Arch setup 🤡"
```

### How to use it

```bash
# Save the script
nano install-arch-clown.sh
# Paste the content above, save & exit

chmod +x install-arch-clown.sh
./install-arch-clown.sh
```

### What the script does
1. Installs all core packages (`i3`, `polybar`, `rofi`, `terminator`, `picom`, `neovim`, fonts, etc.)
2. Installs `yay` if missing
3. Installs Oh My Zsh + Powerlevel10k + the two popular plugins
4. Installs Atuin
5. Clones the official repo and deploys the configs
6. Replaces every occurrence of `changeme` / `/home/changeme` with your real username/home
7. Makes the helper scripts executable
8. Changes your default shell to zsh
